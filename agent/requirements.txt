# NVIDIA's built-in `_type: react_agent` is implemented by the NAT LangChain
# plugin. 1.9.0 splits that plugin's provider integrations into optional
# extras, so this installs only the OpenAI one instead of the complete set
# 1.8.0 hard-depended on (langchain-aws, -community, -exa, -huggingface,
# -litellm, -milvus, -nvidia-ai-endpoints, -oci, plus wikipedia and openevals).
# The core pins are unchanged across the two releases -- langchain >=1.3.3,
# langchain-core >=1.4.0, langgraph >=1.0.5 -- so this is a smaller install of
# the same framework, not a framework migration.
nvidia-nat==1.9.0
nvidia-nat-langchain[openai]==1.9.0
nvidia-nat-opentelemetry==1.9.0

# Required by NAT 1.9's SQLAlchemy asyncio use (the execution and job stores,
# and so the approval interaction path). SQLAlchemy itself is installed
# transitively, but `greenlet` -- which its asyncio module needs -- is declared
# by nothing in this dependency set (resolved for linux/aarch64 and
# linux/x86_64). Without it the agent fails at workflow initialisation with
# "The SQLAlchemy asyncio module requires that the Python 'greenlet' library is
# installed", a startup failure rather than anything a unit test would catch.
# Named as the extra rather than as `greenlet` directly, so the reason travels
# with the requirement.
sqlalchemy[asyncio]

# NeMo Guardrails middleware. This still constrains nemoguardrails to
# >=0.11,<0.22 in 1.9.0 -- the requirement is byte-identical to 1.8.0's -- which
# is why the pin below is 0.21.0 and not the current release. Three streaming
# rail defects in 0.21 are fixed in 0.23.0 and are worked around from
# application code in nat_streaming_react.guardrails_compat; that module
# self-disables and can be deleted once this pin allows >=0.23.
nvidia-nat-security[guardrails]==1.9.0

# MCP client integration used to discover the Rust server's tools.
nvidia-nat-mcp==1.9.0
# Guardrails tracing plus deterministic regex output protection. The `sdd`
# (Presidio) extra is intentionally NOT installed: generic NER masking corrupts
# the exact structured evidence a research decision rests on -- ISINs read as
# IBAN_CODE, and expense ratios, fund sizes and scores are the answer rather
# than incidental PII. See the rails block in agent/config.yml. The template
# this application tracks installs `[sdd,tracing]`; that difference is
# deliberate and must survive upstream synchronisation (docs/UPSTREAM.md).
nemoguardrails[tracing]==0.21.0
